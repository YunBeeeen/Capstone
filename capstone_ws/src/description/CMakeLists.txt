cmake_minimum_required(VERSION 3.8)
project(mobile_robot_description)
find_package(ament_cmake REQUIRED)
install(DIRECTORY urdf meshes launch rviz docs config mjcf
  DESTINATION share/${PROJECT_NAME}
)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
add_executable(collision_scene_node src/collision_scene_node.cpp)
target_compile_features(collision_scene_node PUBLIC cxx_std_17)
ament_target_dependencies(collision_scene_node rclcpp moveit_core moveit_msgs sensor_msgs std_msgs urdf srdfdom)
install(TARGETS collision_scene_node DESTINATION lib/${PROJECT_NAME})
ament_package()
